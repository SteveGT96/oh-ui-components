# SelectField

#### Displays a styled Select field component.

<Canvas of={SelectFieldStories.Default} />

## Usage

```typescript
import { SelectField } from "@/lib/components/ui/select-field";
```

```typescript
<SelectField
    fieldName="SelectField"
    label="SelectField"
    onChange={(value) => {}}
    options={[
        { value: "1", label: "Option 1" },
        { value: "2", label: "Option 2" },
        { value: "3", label: "Option 3" },
    ]}
/>
```

## Props

```typescript
type SelectOption = {
  label: string;
  value: string;
};

interface SelectFieldProps {
  fieldName: string;
  label: string;
  placeholder?: string;
  options: SelectOption[];
  isValid?: boolean;
  fieldValue?: string;
  required?: boolean;
  disabled?: boolean;
  errorText?: string;
  onChange: (value: string) => void;
}
```
