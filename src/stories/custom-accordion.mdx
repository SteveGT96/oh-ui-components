## Usage

```typescript
import { CustomAccordion } from "@/lib/components/ui/custom-accordion";
```

```typescript
<CustomAccordion
    title="Accordion"
    content={<p>Content</p>}
/>
```

## Props

```typescript
interface CustomAccordionProps {
  title: string;
  collapsible?: boolean;
  content: ReactNode;
}
```
